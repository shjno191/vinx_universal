	/*

	function preKeydown() {

		switch (event.keyCode) {
			case 13: // Enter
				// 発注数量欄でEnter押下時に限り、tab遷移動作を実現する為に、Enter自動submit機能回避用に"tab"をセット。
				var syohin_cd = elements_getElementById('syohin_cd');

				}
				break;
			case 121 : // F10 スキャナー取込のトリガーキーコードと重複する為
				return false;
		}

		return true;
	}

	function chkHachuTani(arg) {
		var targetName = arg.name;

		var targetRowIndex = targetName.split("[")[1].split("]")[0];    // RowIndex部分を抽出 meisai_result_rows[0].meisai_nohin_cols[1].headerhachutani → 0

		var headerHachuTani = elements_getElementById(rowsName + "[" + targetRowIndex + "]." + colsName + "[" + targetColIndex + "].headerhachutani").innerText;

		}
	}

	/*
	 * キーダウン
	 */
	function keyDown(arg) {

		var kc = window.event.keyCode;

		if (event.keyCode == 32) {
			if (window.event.srcElement.tagName == 'IMG' && window.event.srcElement.onclick != null) {
				window.event.srcElement.click();
				return false;
			}
		}

		switch (kc) {
			case 13: // Enter
				// 発注数量欄でEnter押下時に限り、tab遷移動作を実現する為に、Enter自動submit機能回避用に"tab"をセット。
				elements_getElementById('pre_method_name').value = "tab";
				moveFocus(arg, 'NEXT');
				break;

			case 38: // ↑
				// 発注窓がフォーカスを失う前に当ロジックに入る為、ここでも発単へ変換

				if (elements_getElementById('syohin_prev_button').disabled) {
					elements_getElementById('update_button').click();
				} else {
					elements_getElementById('syohin_prev_button').click();
				}
				break;


			case 40: // ↓
				// 発注窓がフォーカスを失う前に当ロジックに入る為、ここでも発単へ変換

				if (elements_getElementById('syohin_next_button').disabled) {
					elements_getElementById('update_button').click();
				} else {
					elements_getElementById('syohin_next_button').click();
				}
				break;

			default:
		}
	}

	/*
	 * 登録数欄の間でフォーカスをくるくる遷移させる。
	 * [0-0][0-1][0-2][0-3]
	 * [1-0][1-1][1-2][1-3]
	 */
	function moveFocus(arg, mode) {

		var targetName = arg.name;

		var targetRowIndex = targetName.split("[")[1].split("]")[0];    // RowIndex部分を抽出 meisai_result_rows[0].meisai_nohin_cols[1].headerhachutani → 0
		var targetColIndex = targetName.split("[")[2].split("]")[0];    // ColIndex部分を抽出 meisai_result_rows[0].meisai_nohin_cols[1].headerhachutani → 1

		if (mode == 'NEXT') {
			// 次の入力欄へカーソル遷移
			setFocusNext(targetRowIndex, targetColIndex);
		} else {
			// 前の入力欄へカーソル遷移
			setFocusPrev(targetRowIndex, targetColIndex);
		}
	}

	/*
	 * ↓ボタン押下時、Tab 遷移状態を再現
	 */
	function setFocusNext(targetRowIndex, targetColIndex) {

		targetRowIndex = parseInt(targetRowIndex, 10);
		targetColIndex = parseInt(targetColIndex, 10);

		// key押下時から1つ進んだ入力位置を計算
		if (targetColIndex == 3) {
			targetColIndex = 0;
			if (targetRowIndex == 1) {
				targetRowIndex = 0;
			} else {
				targetRowIndex = targetRowIndex + 1;
			}
		} else {
			targetColIndex = targetColIndex + 1;
		}

		// 入力欄が非表示の場合は、フォーカスセット可能な入力欄を探す
		for (rowNext = targetRowIndex; rowNext < 2; rowNext++) {
			for (colNext = targetColIndex; colNext < 4; colNext++) {
				var hachuSuryoQt = elements_getElementById("rows[" + String(rowNext) + "].cols[" + String(colNext) + "].hachu_suryo_qt.value");
				if (hachuSuryoQt.style.display != 'none') {

					// 何らかのkey押下時にはonblurイベントが発生しているが、
					// その際、focus設定が正常に動作しない。
					// setTimeoutで回避している。。
					//setTimeout(function(){
					//  hachuSuryoQt.focus();
					//}, 1);
					return;
				}
			}

			// 列位置を左端に戻す
			targetColIndex = 0;

			if (rowNext == 1) {
				// 行位置を１行目に戻す
				rowNext = -1; // 次の増分で0となる
			}
		}
	}

	/*
	 * ↑ボタン押下時、Shift + Tab 遷移状態を再現
	 */
	function setFocusPrev(targetRowIndex, targetColIndex) {

		targetRowIndex = parseInt(targetRowIndex, 10);
		targetColIndex = parseInt(targetColIndex, 10);

		if (targetColIndex == 0) {
			targetColIndex = 3;
			if (targetRowIndex == 0) {
				targetRowIndex = 1;
			} else {
				targetRowIndex = targetRowIndex - 1;
			}
		} else {
			targetColIndex = targetColIndex - 1;
		}

		for (rowPrev = targetRowIndex; rowPrev > -1; rowPrev--) {
			for (colPrev = targetColIndex; colPrev > -1; colPrev--) {
				var hachuSuryoQt = elements_getElementById("rows[" + String(rowPrev) + "].cols[" + String(colPrev) + "].hachu_suryo_qt.value");
				if (hachuSuryoQt.style.display != 'none') {
					//setTimeout(function(){hachuSuryoQt.focus();}, 1);
					return;
				}
			}
			targetColIndex = 3;
			if (rowPrev == 0) {
				rowPrev = 2;
			}
		}
	}
